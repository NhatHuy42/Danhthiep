    <script>
        // TYPEWRITER EFFECT
        const roles = ["Học Sinh 11 Toán 2 Chuyên Bến Tre", "Tư Duy Thủ Lĩnh & Sáng Tạo", "Full-Stack Developer & Logic Enthusiast"];
        let rIdx = 0, cIdx = 0, isDeleting = false;
        function typeEffect() {
            const current = roles[rIdx];
            const el = document.getElementById('typewriter');
            if (!el) return;
            if (isDeleting) {
                el.innerText = current.substring(0, cIdx--);
                if (cIdx < 0) { isDeleting = false; rIdx = (rIdx + 1) % roles.length; setTimeout(typeEffect, 400); return; }
            } else {
                el.innerText = current.substring(0, cIdx++);
                if (cIdx > current.length) { isDeleting = true; setTimeout(typeEffect, 1800); return; }
            }
            setTimeout(typeEffect, isDeleting ? 40 : 80);
        }
        document.addEventListener('DOMContentLoaded', typeEffect);

        // CANVAS BACKGROUND ANIMATION
        const canvas = document.getElementById('math-canvas');
        const ctx = canvas.getContext('2d');
        let pts = [];
        function resizeCanvas() { canvas.width = window.innerWidth; canvas.height = window.innerHeight; }
        window.addEventListener('resize', resizeCanvas);
        resizeCanvas();

        for(let i=0; i<45; i++) {
            pts.push({
                x: Math.random()*canvas.width,
                y: Math.random()*canvas.height,
                vx: (Math.random()-0.5)*0.6,
                vy: (Math.random()-0.5)*0.6
            });
        }

        function drawCanvas() {
            ctx.clearRect(0,0,canvas.width,canvas.height);
            ctx.fillStyle = 'rgba(99, 102, 241, 0.4)';
            ctx.strokeStyle = 'rgba(99, 102, 241, 0.08)';

            pts.forEach((p, idx) => {
                p.x += p.vx; p.y += p.vy;
                if(p.x<0 || p.x>canvas.width) p.vx *= -1;
                if(p.y<0 || p.y>canvas.height) p.vy *= -1;

                ctx.beginPath();
                ctx.arc(p.x, p.y, 2, 0, Math.PI*2);
                ctx.fill();

                for(let j=idx+1; j<pts.length; j++) {
                    let p2 = pts[j];
                    let dist = Math.hypot(p.x-p2.x, p.y-p2.y);
                    if(dist < 130) {
                        ctx.beginPath();
                        ctx.moveTo(p.x, p.y);
                        ctx.lineTo(p2.x, p2.y);
                        ctx.stroke();
                    }
                }
            });
            requestAnimationFrame(drawCanvas);
        }
        drawCanvas();

        }
